CREATE Procedure [dbo].[AuthenticateUser]    
    @UserName NVARCHAR(255),  -- Username or Email for login      
    @Password NVARCHAR(255),  -- Password provided by the user      
    @UserId INT OUTPUT,       -- Output parameter to return the user ID      
    @UserRole NVARCHAR(255) OUTPUT      
as begin     
     DECLARE @StoredPassword NVARCHAR(255);    
  DECLARE @RoleId INT;      
  select @UserId = Id, @StoredPassword = Password from [User] WHERE Email = @UserName;     
    
  IF @UserId IS NULL      
  BEGIN      
        SET @UserRole = 'Invalid user name or password';      
        RETURN;      
    END      
  -- Verify the password      
    IF @StoredPassword <> @Password      
    BEGIN      
        SET @UserRole = 'Invalid user name or password';      
        SET @UserId = NULL;      
        RETURN;      
    END     
    
 SELECT @RoleId = RoleId      
    FROM RoleAssign      
    WHERE UserId = @UserId;      
    
  -- If RoleId is null, return an error      
 IF @RoleId IS NULL      
    BEGIN      
        SET @UserRole = 'Role not assigned';      
        RETURN;      
    END     
     -- Fetch the role name from the Roles table using RoleId      
  SELECT @UserRole = Title      
    FROM [Role]      
    WHERE Id = @RoleId;     
    
end
GO

CREATE PROCEDURE [dbo].[GetKeys]        
AS        
BEGIN        
    SET NOCOUNT ON;        
        
    BEGIN TRY        
        SELECT         
            Id,         
            Name,         
            Type,  
   SVG_Value,  
            Description,         
            CreatedOn        
        FROM         
            [Key]  -- Assuming 'InfoKeys' is the table name        
        ORDER BY         
            [Type], CreatedOn DESC;  -- Sorting by latest created keys        
    END TRY        
    BEGIN CATCH        
        -- Error handling        
        DECLARE @ErrorMessage NVARCHAR(4000), @ErrorSeverity INT, @ErrorState INT;        
        SELECT         
            @ErrorMessage = ERROR_MESSAGE(),         
            @ErrorSeverity = ERROR_SEVERITY(),         
            @ErrorState = ERROR_STATE();        
                
        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);        
    END CATCH        
END;     
GO

CREATE PROCEDURE [dbo].[sp_UpInsertWebInfo]          
    @Id INT,          
    @Name NVARCHAR(200),          
    @Type NVARCHAR(100),      
 @SVG NVARCHAR(Max) = Null,      
    @Description NVARCHAR(MAX),          
    @Result INT OUTPUT          
AS          
BEGIN          
    SET NOCOUNT ON;          
          
    BEGIN TRY          
        -- Check if record exists          
        IF EXISTS (SELECT 1 FROM [Key] WHERE Id = @Id)          
        BEGIN          
            -- Update existing record          
            UPDATE [Key]          
            SET           
                Name = @Name,          
                Type = @Type,    
                SVG_Value = @SVG,    
                Description = @Description          
            WHERE Id = @Id;          
          
            SET @Result = @Id; -- Return same Id to indicate update          
        END          
        ELSE          
        BEGIN          
            -- Insert new record          
            INSERT INTO [Key] (Name, Type,SVG_Value, Description, CreatedOn)          
            VALUES (@Name, @Type, @SVG, @Description, SYSUTCDATETIME());          
          
            SET @Result = SCOPE_IDENTITY(); -- Return new Id          
        END          
    END TRY          
    BEGIN CATCH          
        SET @Result = 0; -- Error occurred          
    END CATCH          
END
