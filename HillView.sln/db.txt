CREATE Procedure [dbo].[AuthenticateUser]    
    @UserName NVARCHAR(255),  -- Username or Email for login      
    @Password NVARCHAR(255),  -- Password provided by the user      
    @UserId INT OUTPUT,       -- Output parameter to return the user ID      
    @UserRole NVARCHAR(255) OUTPUT      
as begin     
     DECLARE @StoredPassword NVARCHAR(255);    
  DECLARE @RoleId INT;      
  select @UserId = Id, @StoredPassword = Password from [User] WHERE Email = @UserName;     
    
  IF @UserId IS NULL      
  BEGIN      
        SET @UserRole = 'Invalid user name or password';      
        RETURN;      
    END      
  -- Verify the password      
    IF @StoredPassword <> @Password      
    BEGIN      
        SET @UserRole = 'Invalid user name or password';      
        SET @UserId = NULL;      
        RETURN;      
    END     
    
 SELECT @RoleId = RoleId      
    FROM RoleAssign      
    WHERE UserId = @UserId;      
    
  -- If RoleId is null, return an error      
 IF @RoleId IS NULL      
    BEGIN      
        SET @UserRole = 'Role not assigned';      
        RETURN;      
    END     
     -- Fetch the role name from the Roles table using RoleId      
  SELECT @UserRole = Title      
    FROM [Role]      
    WHERE Id = @RoleId;     
    
end
GO

CREATE   PROCEDURE [dbo].[sp_GetHomePageData]
    @PageId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    --------------------------------------------------
    -- Result Set 1: Active page sections
    --------------------------------------------------
SELECT
    ps.Id AS PageSectionId,
    ps.PageId,
    p.Title AS PageTitle,  -- Add this line
    ps.SectionId,
    ps.DisplayOrder AS PageSectionOrder,
    s.SectionTypeId,
    s.Title,
    s.Heading,
    s.SubHeading,
    s.Description,
    s.InternalName,
    s.BackgroundColor,
    s.CssClass,
    s.SettingsJson,
    s.Status
FROM dbo.PageSections ps
INNER JOIN dbo.Pages p ON p.Id = ps.PageId
INNER JOIN dbo.Sections s ON s.Id = ps.SectionId
WHERE ps.PageId = @PageId
  AND p.Status = 1
  AND ps.Status = 1
  AND s.Status = 1
ORDER BY ps.DisplayOrder;

    --------------------------------------------------
    -- Result Set 2: Active section items
    --------------------------------------------------
    SELECT
        si.Id,
        si.SectionId,
        si.ParentSectionItemId,
        si.Title,
        si.Heading,
        si.SubHeading,
        si.Description,
        si.Value,
        si.Label,
        si.LinkUrl,
        si.AdditionalData,
        si.DisplayOrder,
        si.Status
    FROM dbo.SectionItems si
    INNER JOIN dbo.PageSections ps
        ON ps.SectionId = si.SectionId
    INNER JOIN dbo.Pages p
        ON p.Id = ps.PageId
    INNER JOIN dbo.Sections s
        ON s.Id = ps.SectionId
    WHERE ps.PageId = @PageId
      AND p.Status = 1
      AND ps.Status = 1
      AND s.Status = 1
      AND si.Status = 1
    ORDER BY
        si.SectionId,
        si.ParentSectionItemId,
        si.DisplayOrder;

    --------------------------------------------------
    -- Result Set 3: Active images
    --------------------------------------------------
    SELECT
        i.Id,
        i.SectionId,
        i.SectionItemId,
        i.ImageUrl,
        i.AltText,
        i.Caption,
        i.ImageType,
        i.DisplayOrder,
        i.Status
    FROM dbo.Images i
    INNER JOIN dbo.PageSections ps
        ON ps.SectionId = i.SectionId
    INNER JOIN dbo.Pages p
        ON p.Id = ps.PageId
    INNER JOIN dbo.Sections s
        ON s.Id = ps.SectionId
    WHERE ps.PageId = @PageId
      AND p.Status = 1
      AND ps.Status = 1
      AND s.Status = 1
      AND i.Status = 1
      AND
      (
          i.SectionItemId IS NULL
          OR EXISTS
          (
              SELECT 1
              FROM dbo.SectionItems si
              WHERE si.Id = i.SectionItemId
                AND si.SectionId = i.SectionId
                AND si.Status = 1
          )
      )
    ORDER BY
        i.SectionId,
        i.SectionItemId,
        i.DisplayOrder;

    --------------------------------------------------
    -- Result Set 4: Active buttons
    --------------------------------------------------
    SELECT
        b.Id,
        b.SectionId,
        b.SectionItemId,
        b.Title,
        b.Url,
        b.Target,
        b.CssClass,
        b.ButtonType,
        b.DisplayOrder,
        b.Status
    FROM dbo.Buttons b
    INNER JOIN dbo.PageSections ps
        ON ps.SectionId = b.SectionId
    INNER JOIN dbo.Pages p
        ON p.Id = ps.PageId
    INNER JOIN dbo.Sections s
        ON s.Id = ps.SectionId
    WHERE ps.PageId = @PageId
      AND p.Status = 1
      AND ps.Status = 1
      AND s.Status = 1
      AND b.Status = 1
      AND
      (
          b.SectionItemId IS NULL
          OR EXISTS
          (
              SELECT 1
              FROM dbo.SectionItems si
              WHERE si.Id = b.SectionItemId
                AND si.SectionId = b.SectionId
                AND si.Status = 1
          )
      )
    ORDER BY
        b.SectionId,
        b.SectionItemId,
        b.DisplayOrder;

    --------------------------------------------------
    -- Result Set 5: Active testimonials
    -- Testimonials currently appear only on Home.
    --------------------------------------------------
    SELECT
        t.Id,
        t.ClientName,
        t.ReviewText,
        t.Rating,
        t.DisplayOrder
    FROM dbo.Testimonials t
    WHERE t.Status = 1
    ORDER BY t.DisplayOrder;
END;
GO

CREATE PROCEDURE [dbo].[sp_UpInsertWebInfo]
    @Id INT,
    @Name NVARCHAR(200),
    @Type NVARCHAR(100),
    @SVG NVARCHAR(MAX) = NULL,
    @Description NVARCHAR(MAX),
    @GroupName NVARCHAR(100) = NULL,
    @DisplayOrder INT = 0,
    @Result INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        IF EXISTS (
            SELECT 1
            FROM dbo.[Key]
            WHERE Id = @Id
        )
        BEGIN
            UPDATE dbo.[Key]
            SET
                Name = @Name,
                Type = @Type,
                SVG_Value = @SVG,
                Description = @Description,
                GroupName = @GroupName,
                DisplayOrder = @DisplayOrder
            WHERE Id = @Id;

            SET @Result = @Id;
        END
        ELSE
        BEGIN
            INSERT INTO dbo.[Key]
            (
                Name,
                Type,
                SVG_Value,
                Description,
                CreatedOn,
                GroupName,
                DisplayOrder
            )
            VALUES
            (
                @Name,
                @Type,
                @SVG,
                @Description,
                SYSUTCDATETIME(),
                @GroupName,
                @DisplayOrder
            );

            SET @Result = CONVERT(INT, SCOPE_IDENTITY());
        END
    END TRY
    BEGIN CATCH
        SET @Result = 0;
    END CATCH
END;
GO
